tool.terracek.com {
  # Protect both the browser UI and the API with Basic Auth.
  @hakolect_protected path /hakolect/* /api/hakolect/*
  basicauth @hakolect_protected {
    kiyopi <hashed_password>
  }

  # API routes for external clients (OpenClaw, Chrome extension)
  # Requests must also include X-API-Key: <value from API_KEY env var>.
  handle /api/hakolect/* {
    reverse_proxy 127.0.0.1:8000
  }

  # Web UI - served by the frontend container through the local host port.
  handle /hakolect/* {
    reverse_proxy 127.0.0.1:3000
  }

  # Optional convenience redirect.
  handle = /hakolect {
    redir /hakolect/ 308
  }
}

# Notes:
# - basicauth uses Caddy's hashed password format. Generate with:
#     caddy hash-password
# - Browser access uses Basic Auth only.
# - External API clients must send both Basic Auth and:
#     X-API-Key: <value from API_KEY env var>
# - docker-compose.yml binds the app ports to 127.0.0.1 only so the app is
#   reachable through Caddy but not exposed directly on the public interface.
# - The DNS A record for tool.terracek.com pointing to the VPS is managed
#   outside this codebase.
